# Пункт аудита 30.2, недоработка Б: «Уступ на стене» — ступенчатые помещения
Исполнитель: sol / high

Цель: в разделе «Форма» боковой панели помещения появляется действие «Уступ на стене»: стена разбивается в выбранном месте, одна её часть утапливается внутрь помещения на заданную глубину. Повторяя действие, можно получить «лесенку», Z-форму и т. п.

Правки — только в `index.html`.

## 1. Слой стен: функция `stepRoomWall(room, id, off, h, far)`
Добавить рядом с `setRoomShape` (после неё). Возвращает `{ok:true, moved}` или `{error:'…'}`; при ошибке помещение не меняется.

- `id` — id стены, `off` — место уступа в отсчёте `off` этой стены (от угла, который называет `wallFrom(room,id)`, как у проёмов), `h` — глубина внутрь (м), `far` — `true`: утопить дальнюю от угла отсчёта часть, `false`: ближнюю.
- Вершины: как в `wallDragStart` — `P = roomPoly(room).map((q,i)=>({...q, wid:q.wid||walls[i].id}))`, где `walls=roomWalls(room)`; стена `w` с индексом `i` идёт от `a=P[i]` к `b=P[(i+1)%n]`, `u=w.dir`, нормаль внутрь `nv={x:w.nx,y:w.ny}`.
- Точка разрыва: `t = w.offFrom==='start' ? off : w.len-off` (расстояние от `a`), `s = a + u·t`. Требовать `0.3 <= t <= w.len-0.3`, иначе ошибка «Уступ слишком близко к углу — от угла не меньше 300 мм».
- Какая часть утапливается: «дальняя» = хвост `s..b`, если `w.offFrom==='start'`, иначе голова `a..s`; `far===false` — наоборот.
  - хвост внутрь: вместо `a, b` получить `a, s, s+nv·h, b+nv·h` (у `s` — без `wid`, у `s+nv·h` — без `wid`, вершина `b+nv·h` сохраняет `wid` вершины `b`);
  - голова внутрь: `a+nv·h` (сохраняет `wid` вершины `a`), `s+nv·h` (без `wid`), `s` (новая стена от `s` к `b` получает `wid` стены `w`, т. е. у вершины `s` `wid = w.id`), затем `b`.
  - В варианте «хвост» стена от `a` до `s` сохраняет `wid` стены `w` (вершина `a` уже его несёт).
  - Соседняя стена со стороны сдвигаемого конца должна быть перпендикулярна `w`; если нет (`|u·dir соседа| > 1e-9`) — ошибка «Уступ рядом с косой стеной пока не поддерживается».
  - Массив вершин — всегда новый; `room.poly` на месте не менять (кэш стен сравнивает ссылку `room.poly`).
- Проверки до изменений:
  - `shapeError(normPoly(pts, pts.map(p=>p.wid||null)).pts)` — при ошибке вернуть `{error:'Уступ не помещается: '+текст}`;
  - точка разрыва внутри проёма (`room.openings` с `side===id`, интервал `(o.off, o.off+o.w)`) или вытяжки (`room.points` с `type==='hood'`, `(p.off, p.off+p.w)`) — ошибка «Уступ приходится на „<opName(o) или Зонт>“ — сдвиньте место уступа». Точку без ширины на самом месте разрыва (|p.off-off|<1e-6) тоже считать помехой.
- Перенос объектов с утапливаемой части: до изменения для каждого проёма и точки на стене `id`, лежащих целиком на утапливаемой части, запомнить точку `q = wallPoint(room, id, o.off)` (для точек — `p.off`) и ширину. Затем `const r=setRoomShape(room,{tpl:'custom',pts})`. Найти новую стену Y: направление совпадает с `u`, на её отрезке лежит точка `s + nv·h + u·(±0.01)` (знак — в сторону утопленной части). Каждому запомненному объекту: `side=Y.id`, `off=projectToWall(room,Y.id,q.x+nv.x*h,q.y+nv.y*h).off`, вернуть прежнюю ширину; затем `clampRoom({room,openings:room.openings,points:room.points})`. Вернуть `{ok:true, moved:r.moved}`.
- `setRoomShape`, `assignWids`, `shapeError`, `normPoly`, `clampRoom` — не менять.

function shapeError(pts){
  const P=Array.isArray(pts)?pts:[], n=P.length;
  if(n<4||P.some(p=>!Number.isFinite(p.x)||!Number.isFinite(p.y))) return 'Контур не замкнут';
  let area2=0; for(let i=0;i<n;i++){ const a=P[i], b=P[(i+1)%n]; area2+=a.x*b.y-b.x*a.y; }
  if(Math.abs(area2)<=1e-12) return 'Контур не замкнут';
  const walls=[];
  for(let i=0;i<n;i++){ const a=P[i], b=P[(i+1)%n], dx=b.x-a.x, dy=b.y-a.y;
    if(Math.abs(dx)>1e-9&&Math.abs(dy)>1e-9) return `Стена ${i+1} — не под прямым углом`;
    walls.push({a,b,dx,dy,len:Math.hypot(dx,dy)}); }
  for(let i=0;i<n;i++) if(walls[i].len<0.3) return `Стена ${i+1} короче 300 мм`;
  const cross=(a,b,c)=>(b.x-a.x)*(c.y-a.y)-(b.y-a.y)*(c.x-a.x), on=(a,b,p)=>p.x>=Math.min(a.x,b.x)-1e-9&&p.x<=Math.max(a.x,b.x)+1e-9&&p.y>=Math.min(a.y,b.y)-1e-9&&p.y<=Math.max(a.y,b.y)+1e-9;
  const intersects=(a,b,c,d)=>{ const ab1=cross(a,b,c), ab2=cross(a,b,d), cd1=cross(c,d,a), cd2=cross(c,d,b);
    if(((ab1>1e-9&&ab2< -1e-9)||(ab1< -1e-9&&ab2>1e-9))&&((cd1>1e-9&&cd2< -1e-9)||(cd1< -1e-9&&cd2>1e-9))) return true;
    return (Math.abs(ab1)<=1e-9&&on(a,b,c))||(Math.abs(ab2)<=1e-9&&on(a,b,d))||(Math.abs(cd1)<=1e-9&&on(c,d,a))||(Math.abs(cd2)<=1e-9&&on(c,d,b)); };
  for(let i=0;i<n;i++) for(let j=i+1;j<n;j++){ if(j===i+1||(i===0&&j===n-1)) continue;
    if(intersects(walls[i].a,walls[i].b,walls[j].a,walls[j].b)) return `Стены ${i+1} и ${j+1} пересекаются`; }
  for(let i=0;i<n;i++) for(let j=i+1;j<n;j++){ const a=walls[i], b=walls[j], ux=a.dx/a.len, uy=a.dy/a.len, nx=-uy, ny=ux;
    const vx=b.dx/b.len, vy=b.dy/b.len; if(nx*(-vy)+ny*vx>-1+1e-9) continue;
    const dist=(b.a.x-a.a.x)*nx+(b.a.y-a.a.y)*ny; if(dist<=1e-9||dist>=0.6) continue;
    const q1=(b.a.x-a.a.x)*ux+(b.a.y-a.a.y)*uy, q2=(b.b.x-a.a.x)*ux+(b.b.y-a.a.y)*uy;
    if(Math.min(a.len,Math.max(q1,q2))-Math.max(0,Math.min(q1,q2))>1e-9) return `Между стенами ${i+1} и ${j+1} меньше 600 мм`; }
  return null; }

  for(const p of room.points) if(p.side!=='floor'){ if(newIds.has(p.side)){ const before=oldWalls.find(w=>w.id===p.side), after=newById[p.side];
      if(before&&after&&before.axis===after.axis&&Math.abs(before.nx-after.nx)<1e-9&&Math.abs(before.ny-after.ny)<1e-9){ const q=wallPoint(old,p.side,Number(p.off)||0); p.off=projectToWall(room,p.side,q.x-shiftX,q.y-shiftY).off; }

  return {ok:true,moved}; }

function mirrorRoom(s){ const W=s.room.w, R=s.room;

function clampRoom(s){
  for(const o of s.openings){ const L=sideLen(s,o.side); o.w=Math.min(Math.max(o.type==='prot'?0.05:0.3,o.w),L); o.off=Math.min(Math.max(0,o.off),L-o.w); if(o.type==='prot') o.depth=Math.min(Math.max(0.05,o.depth||0.3),o.niche?3:wallInward(s.room,o.side,o.off,o.w)-0.5); }
  for(const p of s.points){
    if(!p.side) migratePoint(s,p);
    const hood=p.type==='hood';
    if(p.side!=='floor'){ const L=sideLen(s,p.side);
      if(hood){ p.w=Math.min(Math.max(0.3,p.w||1.5),L); p.d=Math.min(Math.max(0.3,p.d||1),wallInward(s.room,p.side,p.off,p.w)); p.off=Math.min(Math.max(0,p.off),L-p.w); }
      else p.off=Math.min(Math.max(0.05,p.off),L-0.05); }
    else if(hood){ p.w=Math.min(Math.max(0.3,p.w||1.5),s.room.w); p.d=Math.min(Math.max(0.3,p.d||1),s.room.d); p.x=Math.min(Math.max(0,p.x),s.room.w-p.w); p.y=Math.min(Math.max(0,p.y),s.room.d-p.d); }
    else { p.x=Math.min(Math.max(0.05,p.x),s.room.w-0.05); p.y=Math.min(Math.max(0.05,p.y),s.room.d-0.05); }
  }
}
